cmake_minimum_required(VERSION 3.21)
project(l2_d3d9_proxy VERSION 1.4.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "This C4 proxy requires Windows x86. Use -A Win32 with Visual Studio.")
endif()
if(NOT MSVC)
    message(FATAL_ERROR "This build is validated with MSVC/Windows SDK. MinGW is not currently supported.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_compile_options(/W4 /WX /permissive- /utf-8 /EHsc)
# GetProcAddress is the documented Windows function-pointer conversion.
add_compile_options(/wd4191)
add_library(d3d9 SHARED d3d9_l2.cpp proxy_support.h d3d9.def)
target_link_libraries(d3d9 PRIVATE dxguid user32)
set_target_properties(d3d9 PROPERTIES PREFIX "" OUTPUT_NAME "d3d9")
include(CTest)
if(BUILD_TESTING)
    add_executable(proxy_tests tests/proxy_tests.cpp)
    target_link_libraries(proxy_tests PRIVATE dxguid user32)
    add_test(NAME policy_and_config COMMAND proxy_tests --unit)
    add_test(NAME native_d3d9 COMMAND proxy_tests --native)
    set_tests_properties(native_d3d9 PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
    add_executable(dll_smoke tests/dll_smoke.cpp)
    target_link_libraries(dll_smoke PRIVATE dxguid user32)
    add_test(NAME dll_exports_and_load COMMAND dll_smoke "$<TARGET_FILE:d3d9>")
    set_tests_properties(dll_exports_and_load PROPERTIES TIMEOUT 30)
endif()
install(TARGETS d3d9 RUNTIME DESTINATION .)
install(FILES config/c4/d3d9_proxy.ini DESTINATION .)
